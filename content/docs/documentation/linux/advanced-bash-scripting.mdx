## Advanced Script Structure

### Setting Up Script Foundations

```bash
#!/bin/bash
set -euo pipefail  # Exit on error, undefined vars, pipe failures
IFS=$'\n\t'        # Better word splitting

# Script metadata
readonly SCRIPT_NAME=$(basename "$0")
readonly SCRIPT_DIR=$(cd "$(dirname "$0")" && pwd)
readonly VERSION="1.0.0"

# Logging function
log() {
    echo "[$(date +'%Y-%m-%d %H:%M:%S')] $*" >&2
}

error() {
    log "ERROR: $*"
    exit 1
}

# Main function
main() {
    log "Starting $SCRIPT_NAME v$VERSION"
    # Your code here
}

# Run main
main "$@"
```

## Working with Arrays

### Basic Array Operations

````bash
# Declare array

# Iterate over array
for server in "${servers[@]}"; do
````

## Error Handling

### Using Traps

```bash
# Trap errors
trap 'error "Script failed at line $LINENO"' ERR

# Cleanup on exit
cleanup() {
    log "Cleaning up..."
    rm -f /tmp/temp-file-*
}
trap cleanup EXIT
```

### Implementing Retry Logic

```bash
# Retry logic
retry() {
    local max_attempts=3
    local attempt=1
    local delay=5

    while [ $attempt -le $max_attempts ]; do
        if "$@"; then
            return 0
        fi

        log "Attempt $attempt failed. Retrying in ${delay}s..."
        sleep "$delay"
        ((attempt++))
    done

    error "Command failed after $max_attempts attempts"
}

# Usage
retry curl -f https://api.example.com/health
```

## Functions and Libraries

### Creating Reusable Functions

```bash
# Function with return value
check_service() {
    local service=$1

    if systemctl is-active --quiet "$service"; then
        echo "running"
        return 0
    else
        echo "stopped"
        return 1
    fi
}

# Usage
status=$(check_service nginx)
if [ "$status" = "running" ]; then
    log "Nginx is running"
fi
```

## Parallel Execution

### Running Tasks Concurrently

```bash
# Run tasks in parallel
run_parallel() {
    local max_jobs=4
    local job_count=0

    for server in "${servers[@]}"; do
        {
            log "Processing $server"
            ssh "$server" "uptime"
        } &

        ((job_count++))

        # Wait if we've reached max parallel jobs
        if (( job_count >= max_jobs )); then
            wait -n  # Wait for any job to finish
            ((job_count--))
        fi
    done

    wait  # Wait for all remaining jobs
}
```

## Configuration Files

### Reading INI-Style Configuration

```bash
# Read INI-style config
read_config() {
    local config_file=$1
    local section=$2
    local key=$3

    awk -F '=' \
        -v section="$section" \
        -v key="$key" \
        '
        $0 ~ "^\\[" section "\\]" { in_section=1; next }
        $0 ~ "^\\[" { in_section=0; next }
        in_section && $1 == key { print $2; exit }
        ' "$config_file" | tr -d ' '
}

# Usage
db_host=$(read_config app.ini database host)
```

## Practical Example: Backup Script

### Complete Backup Implementation

```bash
#!/bin/bash
set -euo pipefail

readonly BACKUP_DIR="/backup"
readonly LOG_FILE="/var/log/backup.log"
readonly RETENTION_DAYS=7

log() {
    echo "[$(date +'%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

create_backup() {
    local source=$1
    local name=$(basename "$source")
    local timestamp=$(date +%Y%m%d_%H%M%S)
    local backup_file="${BACKUP_DIR}/${name}_${timestamp}.tar.gz"

    log "Creating backup of $source"

    if tar -czf "$backup_file" "$source" 2>/dev/null; then
        log "Backup created: $backup_file"
        return 0
    else
        log "ERROR: Backup failed for $source"
        return 1
    fi
}

cleanup_old_backups() {
    log "Removing backups older than $RETENTION_DAYS days"
    find "$BACKUP_DIR" -name "*.tar.gz" -mtime +$RETENTION_DAYS -delete
}

main() {
    log "=== Backup started ==="

    # Create backups
    create_backup "/etc"
    create_backup "/var/www"

    # Cleanup
    cleanup_old_backups

    log "=== Backup completed ==="
}

main "$@"
```
